ARR_Militia_decisions = {
    ARR_spawning_Militias = {
	
	icon = generic_prepare_civil_war
	
        state_target = yes
        target_array = core_states 

        target_trigger = {
            FROM = {
                is_core_of = ROOT
                is_controlled_by = ROOT
                NOT = { has_state_flag = ARR_militia_already_spawned }
            }
        }

        allowed = {
            original_tag = ARR
        }

        visible = {
            has_completed_focus = AAR_An_Militia_Army
			has_war = yes
        }

        cost = 50

        ai_will_do = {
            factor = 50
        }

        fire_only_once = no
        cancel_if_not_visible = yes

        complete_effect = {
            FROM = {
                set_state_flag = ARR_militia_already_spawned
                
                create_unit = {
                    division = "name = \"Militia Division\" division_template = \"Militia\" start_experience_factor = 0.1"
                    owner = ROOT
                }
            }
        }
    }
}
