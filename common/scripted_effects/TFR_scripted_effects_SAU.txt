SAU_update_vision_2030_modifier = {
    
	set_temp_variable = { v2030_scale = SAU_agenda_2030_progress }
    subtract_from_temp_variable = { v2030_scale = 50 }

    # 2. Consumer Goods Factor (von +15% Debuff bei 0% bis -15% Buff bei 100%)
    # Achtung: Bei Consumer Goods ist ein negativer Wert ein Buff!
    set_variable = { SAU_v2030_consumer_goods = v2030_scale }
    multiply_variable = { SAU_v2030_consumer_goods = -0.003 } # -50 * -0.003 = +0.15 | +50 * -0.003 = -0.15

    # 3. Political Power Gain (von -0.25 bei 0% bis +0.25 bei 100%)
    set_variable = { SAU_v2030_pp_gain = v2030_scale }
    multiply_variable = { SAU_v2030_pp_gain = 0.005 } # -50 * 0.005 = -0.25 | +50 * 0.005 = +0.25

    # 4. Baugeschwindigkeit (von -15% bei 0% bis +15% bei 100%)
    set_variable = { SAU_v2030_construction_speed = v2030_scale }
    multiply_variable = { SAU_v2030_construction_speed = 0.003 }

    # 5. Stabilität (von -10% bei 0% bis +10% bei 100%)
    set_variable = { SAU_v2030_stability = v2030_scale }
    multiply_variable = { SAU_v2030_stability = 0.002 }
	
}
